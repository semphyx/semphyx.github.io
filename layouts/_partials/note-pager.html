{{- /* 前後のページへのリンク。ノートは同じ分類の中で日付の古い順、講義ノートは同じ講義の中で章の順 */ -}} {{- $lecture := eq .Section "lectures" -}} {{- $pages := cond $lecture .CurrentSection.RegularPages.ByWeight .CurrentSection.RegularPages.ByDate -}} {{- $prev := false -}}{{- $next := false -}}{{- $found := false -}} {{- $prevNo := 0 -}}{{- $nextNo := 0 -}} {{- range $i, $p := $pages -}} {{- if $found -}}{{- if not $next }}{{ $next = $p }}{{ $nextNo = add $i 1 }}{{ end -}} {{- else if eq $p.RelPermalink $.RelPermalink -}}{{- $found = true -}} {{- else -}}{{- $prev = $p -}}{{ $prevNo = add $i 1 }}{{- end -}} {{- end -}} {{- if or $prev $next }}
{{- with $prev }}← {{ if $lecture }}第 {{ $prevNo }} 章{{ else }}前のノート{{ end }}{{ .Title }}{{ end -}}
{{- with $next }}{{ if $lecture }}第 {{ $nextNo }} 章{{ else }}次のノート{{ end }} →{{ .Title }}{{ end -}}
{{- end }}
